## The AuditTrailPage Object

```typescript
// pages/compliance/AuditTrailPage.ts
import { Page, Locator, expect } from '@playwright/test';
import { AuditEntry } from '../../components/AuditEntry';

export class AuditTrailPage {
  readonly page: Page;
  readonly auditTable: Locator;
  readonly exportButton: Locator;
  readonly filterByUser: Locator;
  readonly filterByAction: Locator;

  constructor(page: Page) {
    this.page = page;
    this.auditTable = page.getByRole('table', { name: 'Audit Trail' });
    this.exportButton = page.getByRole('button', { name: 'Export CSV' });
    this.filterByUser = page.getByLabel('Filter by User');
    this.filterByAction = page.getByLabel('Filter by Action');
  }

  async navigateTo(documentId: string) {
    await this.page.goto(`/documents/${documentId}/audit`);
    await expect(this.auditTable).toBeVisible();
  }

  getAllEntries(): Locator {
    return this.auditTable.getByRole('row').filter({ hasText: /Signed|Submitted|Created/ });
  }

  getLatestEntry() {
    return new AuditEntry(this.getAllEntries().first());
  }

  getEntryByUser(email: string) {
    return new AuditEntry(
      this.getAllEntries().filter({ hasText: email }).first()
    );
  }

  async exportToCSV(): Promise<string> {
    const [download] = await Promise.all([
      this.page.waitForEvent('download'),
      this.exportButton.click(),
    ]);
    const path = await download.path();
    return path!;
  }
}
```

## The AuditEntry Component

```typescript
// components/AuditEntry.ts
import { Locator } from '@playwright/test';

export class AuditEntry {
  readonly user: Locator;
  readonly action: Locator;
  readonly reason: Locator;
  readonly timestamp: Locator;
  readonly ipAddress: Locator;
  readonly documentHash: Locator;

  constructor(private readonly row: Locator) {
    this.user = row.getByTestId('audit-user');
    this.action = row.getByTestId('audit-action');
    this.reason = row.getByTestId('audit-reason');
    this.timestamp = row.getByTestId('audit-timestamp');
    this.ipAddress = row.getByTestId('audit-ip');
    this.documentHash = row.getByTestId('audit-hash');
  }

  async toObject() {
    return {
      user: await this.user.textContent(),
      action: await this.action.textContent(),
      reason: await this.reason.textContent(),
      timestamp: await this.timestamp.textContent(),
      ipAddress: await this.ipAddress.textContent(),
      hash: await this.documentHash.textContent(),
    };
  }
}
```

// utils/compliance-reporter.ts
export async function generateComplianceReport(results: TestResult[]) {
  const report = {
    generated: new Date().toISOString(),
    standard: '21 CFR Part 11',
    
    summary: {
      total: results.length,
      passed: results.filter(r => r.status === 'passed').length,
      failed: results.filter(r => r.status === 'failed').length,
      complianceScore: '0%',
    },
    
    requirements: {
      '§11.10(a) - Validation': 'COMPLIANT',
      '§11.10(d) - Access Controls': 'COMPLIANT',
      '§11.10(e) - Audit Trail': 'COMPLIANT',
      '§11.50 - Signature Manifestations': 'COMPLIANT',
      '§11.200 - Electronic Signature Components': 'COMPLIANT',
    },
  };
  
  const passed = report.summary.passed;
  const total = report.summary.total;
  report.summary.complianceScore = `${Math.round((passed / total) * 100)}%`;
  
  return report;
}

## Common Mistakes

> [!WARNING]
> **Mistake 1: Brittle Implementation**  
> **Wrong:** Tightly coupling test data with page interactions.  
> **Correct:** Abstracting data and logic appropriately.  
> **Why:** Prevents tests from breaking when minor details change.

> [!WARNING]
> **Mistake 2: Missing Synchronization**  
> **Wrong:** Relying on implicit speeds or hardcoded sleeps.  
> **Correct:** Using web-first assertions for synchronization.  
> **Why:** Guarantees deterministic execution regardless of environment speed.
