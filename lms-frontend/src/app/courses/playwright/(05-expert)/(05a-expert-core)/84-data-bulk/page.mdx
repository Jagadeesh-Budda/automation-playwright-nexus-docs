## Why Bulk Data Testing?

Enterprise systems face unique challenges:
- Performance degrades with large datasets (100,000 documents)
- Pagination logic only works correctly when there's enough data to paginate
- Sort algorithms may have edge cases at scale

Bulk data tests simulate real-world conditions.

## Bulk Create via API (Fastest Method)

```typescript
import { test, expect } from '@playwright/test';

test('system handles 100 documents without pagination errors', async ({ request, page }) => {
  const headers = { Authorization: `Bearer ${adminToken}` };
  const createdIds: string[] = [];
  
  // Create 100 documents in batches of 10 (avoid overwhelming the API)
  for (let batch = 0; batch < 10; batch++) {
    const batchPromises = Array.from({ length: 10 }, (_, i) =>
      request.post('/api/documents', {
        data: {
          title: `Bulk Test Doc ${batch * 10 + i + 1}`,
          type: 'Report',
          department: 'QA',
          version: '1.0'
        },
        headers,
      }).then(async r => {
        const doc = await r.json();
        return doc.id;
      })
    );
    
    const batchIds = await Promise.all(batchPromises);
    createdIds.push(...batchIds);
  }
  
  expect(createdIds).toHaveLength(100);
  
  // Verify UI handles the large dataset
  await page.goto('/documents');
  
  // Pagination should be visible
  await expect(page.getByRole('navigation', { name: 'Pagination' })).toBeVisible();
  
  // Total count should be at least 100
  const totalText = await page.getByTestId('total-count').textContent();
  expect(parseInt(totalText!)).toBeGreaterThanOrEqual(100);
  
  // Cleanup
  console.log(`Cleaning up ${createdIds.length} test documents...`);
  await Promise.allSettled(
    createdIds.map(id => request.delete(`/api/documents/${id}`, { headers }))
  );
});
```

## Data Seeding with Factories

```typescript
// utils/DataFactory.ts
import { APIRequestContext } from '@playwright/test';

export class DataFactory {
  constructor(
    private readonly request: APIRequestContext,
    private readonly token: string
  ) {}

  private get headers() {
    return { Authorization: `Bearer ${this.token}` };
  }

  async seedDocuments(count: number, overrides: Partial<any> = {}): Promise<string[]> {
    const ids: string[] = [];
    
    for (let i = 0; i < count; i++) {
      const response = await this.request.post('/api/documents', {
        data: {
          title: `Seeded Document ${Date.now()}-${i}`,
          type: 'Protocol',
          department: 'QA',
          version: '1.0',
          ...overrides
        },
        headers: this.headers,
      });
      const doc = await response.json();
      ids.push(doc.id);
    }
    
    return ids;
  }

  async seedApprovedDocuments(count: number): Promise<string[]> {
    const ids = await this.seedDocuments(count, { type: 'SOP' });
    
    // Approve all in parallel
    await Promise.all(
      ids.map(id =>
        this.request.post(`/api/documents/${id}/approve`, {
          data: { reason: 'Factory approval for testing' },
          headers: this.headers,
        })
      )
    );
    
    return ids;
  }

  async cleanup(ids: string[]): Promise<void> {
    await Promise.allSettled(
      ids.map(id => this.request.delete(`/api/documents/${id}`, { headers: this.headers }))
    );
    console.log(`?? Cleaned up ${ids.length} test records`);
  }
}
```

## Using the DataFactory in Tests

```typescript
import { test, expect } from '../../fixtures/baseTest';
import { APIRequestContext } from '@playwright/test';
import { DataFactory } from '../../utils/DataFactory';

test.describe('Pagination with bulk data', () => {
  let factory: DataFactory;
  let seededIds: string[] = [];
  let apiRequestContext: APIRequestContext;
  
  test.beforeAll(async ({ playwright }) => {
    // request is a test-scoped fixture; instantiate context manually in beforeAll
    apiRequestContext = await playwright.request.newContext();
    
    const loginRes = await apiRequestContext.post('/api/auth/login', {
      data: { email: process.env.ADMIN_USER, password: process.env.ADMIN_PASS }
    });
    const { token } = await loginRes.json();
    
    factory = new DataFactory(apiRequestContext, token);
    
    // Seed 50 documents before tests
    seededIds = await factory.seedDocuments(50);
    console.log(`Seeded ${seededIds.length} documents`);
  });
  
  test.afterAll(async () => {
    await factory.cleanup(seededIds);
    await apiRequestContext.dispose();
  });

  test('first page shows exactly 25 results', async ({ page }) => {
    await page.goto('/documents?page=1&limit=25');
    let rows = page.getByRole('row').filter({ hasText: /DOC-/ });
    await expect(rows).toHaveCount(25);
  });

  test('second page shows the next 25 results', async ({ page }) => {
    await page.goto('/documents?page=2&limit=25');
    const rows = page.getByRole('row').filter({ hasText: /DOC-/ });
    await expect(rows).toHaveCount(25);
    
    // Verify it's different data from page 1
    await page.goto('/documents?page=1&limit=25');
    const page1Ids = await page.getByTestId('doc-id').allTextContents();
    await page.goto('/documents?page=2&limit=25');
    const page2Ids = await page.getByTestId('doc-id').allTextContents();
    
    // No overlap between pages
    const overlap = page1Ids.filter(id => page2Ids.includes(id));
    expect(overlap).toHaveLength(0);
  });
});
```

## Common Mistakes

> [!WARNING]
> **Mistake 1: Brittle Implementation**  
> **Wrong:** Tightly coupling test data with page interactions.  
> **Correct:** Abstracting data and logic appropriately.  
> **Why:** Prevents tests from breaking when minor details change.

> [!WARNING]
> **Mistake 2: Missing Synchronization**  
> **Wrong:** Relying on implicit speeds or hardcoded sleeps.  
> **Correct:** Using web-first assertions for synchronization.  
> **Why:** Guarantees deterministic execution regardless of environment speed.
