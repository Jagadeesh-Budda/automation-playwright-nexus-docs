## What is an AST Linter?

An AST Linter is a tool that walks the AST of your code and reports **rule violations** — specific patterns that you've decided are forbidden or required in your codebase.

Unlike ESLint (which is general-purpose), the ANA linter enforces **automation-specific** rules that ESLint doesn't know about.

## Building a Complete AST Linter

```typescript
// utils/AstLinter.ts
import { parse } from '@typescript-eslint/typescript-estree';
import * as fs from 'fs';
import * as path from 'path';
import { glob } from 'glob';

interface Violation {
  rule: string;
  file: string;
  line: number;
  column: number;
  message: string;
  severity: 'error' | 'warning';
}

interface LintRule {
  name: string;
  severity: 'error' | 'warning';
  check: (ast: any, filePath: string) => Violation[];
}

export class AstLinter {
  private rules: LintRule[] = [];
  private violations: Violation[] = [];

  // Register a rule
  addRule(rule: LintRule): this {
    this.rules.push(rule);
    return this;
  }

  // Lint a single file
  lintFile(filePath: string): Violation[] {
    const code = fs.readFileSync(filePath, 'utf-8');
    let ast: any;
    
    try {
      ast = parse(code, { jsx: false, loc: true, range: true });
    } catch (err: any) {
      return [{ 
        rule: 'parse-error', 
        file: filePath, 
        line: 0, 
        column: 0,
        message: `Failed to parse: ${err.message}`,
        severity: 'error'
      }];
    }
    
    const fileViolations: Violation[] = [];
    for (const rule of this.rules) {
      fileViolations.push(...rule.check(ast, filePath));
    }
    return fileViolations;
  }

  // Lint raw code string directly
  lintCode(code: string, filePath: string): Violation[] {
    let ast: any;
    try {
      ast = parse(code, { jsx: false, loc: true, range: true });
    } catch (err: any) {
      return [{ 
        rule: 'parse-error', 
        file: filePath, 
        line: 0, 
        column: 0,
        message: `Failed to parse: ${err.message}`,
        severity: 'error'
      }];
    }
    
    const codeViolations: Violation[] = [];
    for (const rule of this.rules) {
      codeViolations.push(...rule.check(ast, filePath));
    }
    return codeViolations;
  }

  // Lint all files matching a glob pattern
  lintDirectory(pattern: string): Violation[] {
    const files = glob.sync(pattern);
    this.violations = [];
    
    for (const file of files) {
      this.violations.push(...this.lintFile(file));
    }
    
    return this.violations;
  }

  // Print a readable report
  report(): void {
    const errors = this.violations.filter(v => v.severity === 'error');
    const warnings = this.violations.filter(v => v.severity === 'warning');
    
    for (const v of this.violations) {
      const icon = v.severity === 'error' ? '?' : '??';
      console.log(`${icon} [${v.rule}] ${v.file}:${v.line} — ${v.message}`);
    }
    
    console.log(`\n${this.violations.length} problems (${errors.length} errors, ${warnings.length} warnings)`);
    
    if (errors.length > 0) {
      process.exit(1);  // Fail the CI build
    }
  }
}

// -----------------------------------------------------------------
// Built-in Rules
// -----------------------------------------------------------------

export const noHardcodedSleepRule: LintRule = {
  name: 'no-hardcoded-sleep',
  severity: 'error',
  check(ast, filePath) {
    const violations: Violation[] = [];
    walk(ast, (node) => {
      if (isCallTo(node, 'waitForTimeout') || isCallTo(node, 'sleep')) {
        violations.push({
          rule: 'no-hardcoded-sleep',
          file: filePath,
          line: node.loc.start.line,
          column: node.loc.start.column,
          message: 'Hardcoded sleep is prohibited. Use web-first assertions instead.',
          severity: 'error',
        });
      }
    });
    return violations;
  }
};

// Helper: check if a node is a call to a specific method
function isCallTo(node: any, methodName: string): boolean {
  return (
    node.type === 'CallExpression' &&
    node.callee?.type === 'MemberExpression' &&
    node.callee.property?.name === methodName
  );
}

// Helper: recursive AST walker with parent pointer attachment and circular reference protection
function walk(node: any, visitor: (node: any) => void, parent?: any): void {
  if (!node || typeof node !== 'object') return;
  
  if (parent) {
    node.parent = parent;
  }
  
  visitor(node);
  for (const key of Object.keys(node)) {
    if (key === 'parent') continue; // Prevent circular stack overflow
    if (Array.isArray(node[key])) {
      node[key].forEach((child: any) => walk(child, visitor, node));
    } else {
      walk(node[key], visitor, node);
    }
  }
}
```

## Using the Linter in a Script

```typescript
// scripts/lint.ts
import { AstLinter, noHardcodedSleepRule } from '../utils/AstLinter';

const linter = new AstLinter()
  .addRule(noHardcodedSleepRule);

const violations = linter.lintDirectory('tests/**/*.spec.ts');
linter.report();

// Exits with code 1 if any errors found — fails the CI build!
```

```bash
# Run the linter
npx ts-node scripts/lint.ts
```

## Common Mistakes

> [!WARNING]
> **Mistake 1: Brittle Implementation**  
> **Wrong:** Tightly coupling test data with page interactions.  
> **Correct:** Abstracting data and logic appropriately.  
> **Why:** Prevents tests from breaking when minor details change.

> [!WARNING]
> **Mistake 2: Missing Synchronization**  
> **Wrong:** Relying on implicit speeds or hardcoded sleeps.  
> **Correct:** Using web-first assertions for synchronization.  
> **Why:** Guarantees deterministic execution regardless of environment speed.
