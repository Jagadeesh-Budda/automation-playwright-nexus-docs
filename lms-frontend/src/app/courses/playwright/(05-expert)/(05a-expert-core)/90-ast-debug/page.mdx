## When Your AST Rules Produce False Positives

A **false positive** is when the linter flags code as a violation, even though it's actually correct. This erodes trust in the linter and causes engineers to ignore it.

## Debug Step 1: Visualize the AST

Before writing an AST rule, always visualize the tree structure of the code you're targeting.

```typescript
import { parse } from '@typescript-eslint/typescript-estree';

const code = `
  await page.waitForTimeout(3000);
`;

const ast = parse(code, { loc: true, range: true });
console.log(JSON.stringify(ast.body, null, 2));
```

**Online tool:** Use [AST Explorer](https://astexplorer.net/) to interactively visualize any code's AST tree. Select `@typescript-eslint/typescript-estree` as the parser.

## Debug Step 2: Log Node Types During Walking

```typescript
function debugWalk(node: any, depth = 0): void {
  if (!node || typeof node !== 'object' || !node.type) return;
  
  // Print the node type and relevant properties
  const indent = '  '.repeat(depth);
  const info = node.name ? ` (${node.name})` : 
               node.value !== undefined ? ` (${node.value})` : '';
  console.log(`${indent}${node.type}${info}`);
  
  for (const key of Object.keys(node)) {
    if (['loc', 'range', 'parent'].includes(key)) continue;
    if (Array.isArray(node[key])) {
      node[key].forEach((child: any) => debugWalk(child, depth + 1));
    } else if (typeof node[key] === 'object') {
      debugWalk(node[key], depth + 1);
    }
  }
}

// Usage
debugWalk(ast);
```

## Debug Step 3: Unit Test Your Rules

Write unit tests for your AST rules before deploying them:

```typescript
import { test, expect } from '@playwright/test';

// tests/unit/ast-rules.test.ts
import { AstLinter, noHardcodedSleepRule } from '../../utils/AstLinter';

function lint(code: string) {
  const linter = new AstLinter().addRule(noHardcodedSleepRule);
  // Parse the code string directly
  return linter.lintCode(code, 'test.ts');
}

test('detects waitForTimeout violation', () => {
  let violations = lint(`await page.waitForTimeout(3000);`);
  expect(violations).toHaveLength(1);
  expect(violations[0].rule).toBe('no-hardcoded-sleep');
});

test('does NOT flag waitForSelector (different method name)', () => {
  let violations = lint(`await page.waitForSelector('#btn');`);
  expect(violations).toHaveLength(0);
});

test('does NOT flag expect.toPass with polling', () => {
  const violations = lint(`
    await expect(async () => {
      await expect(page.getByText('Done')).toBeVisible();
    }).toPass({ timeout: 30000 });
  `);
  expect(violations).toHaveLength(0);
});
```

## Common Rule Bugs and Fixes

### Bug 1: Rule matches too broadly
```typescript
// ❌ Bug: Flags ANY call to waitForTimeout, even in helper files
if (isMethodCall(node, 'waitForTimeout')) { ... }

// ✅ Fix: Only flag in spec files, not utility files
if (isMethodCall(node, 'waitForTimeout') && filePath.includes('.spec.')) { ... }
```

### Bug 2: Null reference in AST walking
```typescript
// ❌ Bug: node.callee might be null for some CallExpression types
if (node.callee.property.name === 'waitForTimeout') { ... }  // TypeError!

// ✅ Fix: Use optional chaining
if (node.callee?.property?.name === 'waitForTimeout') { ... }
```

### Bug 3: Missing array iteration
```typescript
// ❌ Bug: Only walks the direct children, misses nested nodes
function walk(node: any, visitor: Function) {
  visitor(node);
  for (const key of Object.keys(node)) {
    walk(node[key], visitor);  // Doesn't handle arrays!
  }
}

// ✅ Fix: Handle arrays and prevent circular parent references
function walk(node: any, visitor: Function) {
  if (!node || typeof node !== 'object') return;
  visitor(node);
  for (const key of Object.keys(node)) {
    if (key === 'parent') continue; // Prevent circular stack overflow
    if (Array.isArray(node[key])) {
      node[key].forEach((child: any) => walk(child, visitor));  // ✅
    } else {
      walk(node[key], visitor);
    }
  }
}
```
