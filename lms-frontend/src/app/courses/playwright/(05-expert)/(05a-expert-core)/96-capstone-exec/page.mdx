### `tests/documents/workflow.spec.ts`

```typescript
import { test, expect } from '../../fixtures/baseTest';
import { DocumentFactory } from '../../utils/DataFactory';

test.describe('Document Review Workflow', () => {
  let docId: string;

  test.beforeEach(async ({ documentPage }) => {
    // Create fresh document for each test
    docId = await documentPage.createDocument(DocumentFactory());
  });

  test('full workflow: Draft ? Pending ? Approved', async ({
    documentPage, reviewPage
  }) => {
    // Act: Submit for review
    await documentPage.navigate();
    await documentPage.submitForReview(docId);
    await expect(documentPage.getStatusBadge(docId)).toContainText('Pending Review');

    // Act: Approve
    await reviewPage.navigateTo(docId);
    await reviewPage.submitReview({ decision: 'Approve', comment: 'All sections complete' });
    
    // Assert
    await documentPage.navigate();
    await expect(documentPage.getStatusBadge(docId)).toContainText('Approved');
  });

  test('rejected documents can be resubmitted', async ({ documentPage, reviewPage }) => {
    await documentPage.navigate();
    await documentPage.submitForReview(docId);

    await reviewPage.navigateTo(docId);
    await reviewPage.submitReview({ decision: 'Reject', comment: 'Missing risk assessment' });
    
    await documentPage.navigate();
    await expect(documentPage.getStatusBadge(docId)).toContainText('Rejected');
    
    // The rejected document can be resubmitted for review
    await documentPage.submitForReview(docId);
    await expect(documentPage.getStatusBadge(docId)).toContainText('Pending Review');
  });
});
```

### `tests/api/documents-api.spec.ts`

```typescript
import { test, expect } from '@playwright/test';

test.describe('Documents API', () => {
  let token: string;
  const createdIds: string[] = [];
  
  test.beforeAll(async ({ request }) => {
    let res = await request.post('/api/auth/login', {
      data: { email: process.env.ADMIN_USER, password: process.env.ADMIN_PASS }
    });
    token = (await res.json()).token;
  });

  test.afterAll(async ({ request }) => {
    await Promise.allSettled(
      createdIds.map(id => request.delete(`/api/documents/${id}`, {
        headers: { Authorization: `Bearer ${token}` }
      }))
    );
  });

  test('POST /api/documents creates document with status Draft', async ({ request }) => {
    let res = await request.post('/api/documents', {
      data: { title: 'API Test Doc', type: 'Protocol', department: 'QA' },
      headers: { Authorization: `Bearer ${token}` }
    });
    
    expect(res.status()).toBe(201);
    const doc = await res.json();
    expect(doc.id).toBeDefined();
    expect(doc.status).toBe('Draft');
    createdIds.push(doc.id);
  });

  test('GET /api/documents returns paginated list', async ({ request }) => {
    const res = await request.get('/api/documents?page=1&limit=10', {
      headers: { Authorization: `Bearer ${token}` }
    });
    
    expect(res.status()).toBe(200);
    const body = await res.json();
    expect(body.data).toBeInstanceOf(Array);
    expect(body.pagination.page).toBe(1);
    expect(body.pagination.limit).toBe(10);
  });
});
```

## Common Mistakes

> [!WARNING]
> **Mistake 1: Brittle Implementation**  
> **Wrong:** Tightly coupling test data with page interactions.  
> **Correct:** Abstracting data and logic appropriately.  
> **Why:** Prevents tests from breaking when minor details change.

> [!WARNING]
> **Mistake 2: Missing Synchronization**  
> **Wrong:** Relying on implicit speeds or hardcoded sleeps.  
> **Correct:** Using web-first assertions for synchronization.  
> **Why:** Guarantees deterministic execution regardless of environment speed.
