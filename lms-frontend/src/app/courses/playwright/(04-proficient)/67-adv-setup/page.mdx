## What is Global Setup?

**Global Setup** runs **once** before the entire test suite starts. It's used for expensive, one-time operations like:
- Logging in and saving authentication state
- Seeding a database with test data
- Setting up external services
- Creating test fixtures (files, folders, users)

**Global Teardown** runs **once** after all tests finish to clean up.

## The Global Setup File

```typescript
// global-setup.ts
import { chromium, FullConfig } from '@playwright/test';
import dotenv from 'dotenv';

dotenv.config();

async function globalSetup(config: FullConfig) {
  console.log('?? Running global setup...');
  
  const { baseURL } = config.projects[0].use;
  const browser = await chromium.launch();

  // Create authentication state for each user role
  await setupAuth(browser, baseURL!, 'admin');
  await setupAuth(browser, baseURL!, 'reviewer');
  await setupAuth(browser, baseURL!, 'viewer');

  await browser.close();
  console.log('? Global setup complete');
}

async function setupAuth(browser: any, baseURL: string, role: string) {
  const context = await browser.newContext();
  const page = await context.newPage();
  
  await page.goto(`${baseURL}/login`);
  await page.getByLabel('Email').fill(process.env[`${role.toUpperCase()}_USER`]!);
  await page.getByLabel('Password').fill(process.env[`${role.toUpperCase()}_PASS`]!);
  await page.getByRole('button', { name: 'Sign In' }).click();
  
  // Wait for successful login
  await page.waitForURL(/dashboard/);
  
  // Save the auth state to a file
  await context.storageState({ path: `.auth/${role}.json` });
  await context.close();
  
  console.log(`  ? Auth state saved for: ${role}`);
}

export default globalSetup;
```

## The Global Teardown File

```typescript
// global-teardown.ts
import { FullConfig } from '@playwright/test';
import { PrismaClient } from '@prisma/client';
import * as fs from 'fs';

async function globalTeardown(config: FullConfig) {
  console.log('?? Running global teardown...');
  
  // 1. Clean up test data from database
  const prisma = new PrismaClient();
  await prisma.document.deleteMany({
    where: { title: { startsWith: 'Test-' } }
  });
  await prisma.$disconnect();
  
  // 2. Clean up temp files
  const tempFiles = fs.readdirSync('./test-fixtures/temp');
  for (const file of tempFiles) {
    fs.unlinkSync(`./test-fixtures/temp/${file}`);
  }
  
  console.log('? Teardown complete');
}

export default globalTeardown;
```

## Registering Setup and Teardown

```typescript
// playwright.config.ts
export default defineConfig({
  globalSetup: './global-setup.ts',
  globalTeardown: './global-teardown.ts',
  
  use: {
    // Default auth state (most tests use admin)
    storageState: '.auth/admin.json',
  },
  
  projects: [
    {
      name: 'admin-tests',
      use: { storageState: '.auth/admin.json' },
    },
    {
      name: 'reviewer-tests',
      use: { storageState: '.auth/reviewer.json' },
    },
    {
      name: 'viewer-tests',
      use: { storageState: '.auth/viewer.json' },
    },
    {
      // Setup project: runs first, no storageState
      name: 'setup',
      testMatch: '**/auth.setup.ts',
    },
  ],
});
```

## Project Dependencies (Setup Before Tests)

A more modern approach uses **project dependencies**:

```typescript
// playwright.config.ts
projects: [
  // STEP 1: Always run auth setup first
  {
    name: 'setup',
    testMatch: /.*\.setup\.ts/,
  },
  
  // STEP 2: Run tests only after setup is complete
  {
    name: 'e2e-tests',
    dependencies: ['setup'],  // Won't start until 'setup' project is done
    use: { storageState: '.auth/admin.json' },
  },
],
```

```typescript
// auth.setup.ts — the setup project's test file
import { test as setup } from '@playwright/test';

setup('authenticate as admin', async ({ page }) => {
  await page.goto('/login');
  await page.getByLabel('Email').fill(process.env.ADMIN_USER!);
  await page.getByLabel('Password').fill(process.env.ADMIN_PASS!);
  await page.getByRole('button', { name: 'Sign In' }).click();
  await page.waitForURL(/dashboard/);
  
  await page.context().storageState({ path: '.auth/admin.json' });
});
```

## Common Mistakes

> [!WARNING]
> **Mistake 1: Brittle Implementation**  
> **Wrong:** Tightly coupling test data with page interactions.  
> **Correct:** Abstracting data and logic appropriately.  
> **Why:** Prevents tests from breaking when minor details change.

> [!WARNING]
> **Mistake 2: Missing Synchronization**  
> **Wrong:** Relying on implicit speeds or hardcoded sleeps.  
> **Correct:** Using web-first assertions for synchronization.  
> **Why:** Guarantees deterministic execution regardless of environment speed.
